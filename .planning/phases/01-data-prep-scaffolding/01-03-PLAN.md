<objective>
Produce the per-polygon padded fruit-disease crop dataset that Phase 3's disease classifier trains on. One script reads Kaggle Afzaal's LabelMe polygons from `data/disease/{train,val,test}/*.json`, computes a padded axis-aligned bounding box around each polygon, crops that region from the source image, and writes it into `data/disease_crops/{split}/{canonical_class}/{image_stem}__{polygon_idx}.jpg`. Leaf-disease classes (`powdery_mildew_leaf`, `leaf_spot`, `angular_leafspot`) are routed to `_leaves_excluded/` rather than dropped — segregated but preserved for potential v2 use. Determinism is load-bearing: re-running the script on the same Kaggle data produces byte-identical output JPEGs (DATA-07).

Purpose: Phase 3's YOLO26n-cls classifier consumes `data/disease_crops/` directly via Ultralytics' standard `ImageFolder`-style layout. The polygon→bbox conversion is the single largest source of non-obvious behavior in the data layer (padding ratio, class segregation, split preservation, JPEG encoding determinism), so it lives in its own plan with explicit verification rather than bundled elsewhere.

Output: A populated `data/disease_crops/` tree, a library-grade cropping helper in `src/manifest/crops.py`, and an idempotent CLI at `scripts/build_disease_crops.py` that is provably deterministic.
</objective>

    ```python
    """Polygon-to-padded-bbox cropping helpers for Kaggle Afzaal disease data.

    Pure functions; deterministic given fixed input bytes. Class-route table
    segregates leaf-disease classes into _leaves_excluded/ rather than dropping them.
    """
    from __future__ import annotations
    import json
    import logging
    from dataclasses import dataclass
    from pathlib import Path
    from typing import Iterator, Literal

    from PIL import Image

    # Canonical disease classes -> training destination.
    # "fruit" classes go under data/disease_crops/{split}/{class}/
    # "leaf"  classes go under data/disease_crops/_leaves_excluded/{split}/{class}/
    DiseaseRole = Literal["fruit", "leaf"]
    CLASS_ROLE: dict[str, DiseaseRole] = {
        "anthracnose_fruit_rot": "fruit",
        "blossom_blight":        "fruit",
        "gray_mold":             "fruit",
        "powdery_mildew_fruit":  "fruit",
        "angular_leafspot":      "leaf",
        "leaf_spot":             "leaf",
        "powdery_mildew_leaf":   "leaf",
    }

    PADDING_RATIO = 0.15          # fixed — must not be parameterized (determinism)
    JPEG_QUALITY = 95             # fixed
    JPEG_SUBSAMPLING = 0          # fixed (4:4:4, no chroma subsampling)

    @dataclass(frozen=True)
    class PolygonRecord:
        """One polygon parsed from a LabelMe JSON, with its provenance."""
        source_image_path: Path
        image_stem: str
        polygon_idx: int
        source_class_name: str    # raw label from JSON
        canonical_class: str      # mapped through DISEASE_CLASSES
        points: tuple[tuple[float, float], ...]  # pixel coords
        split: str                # "train" | "val" | "test"
        role: DiseaseRole         # "fruit" | "leaf"

    def iter_kaggle_polygons(disease_root: Path) -> Iterator[PolygonRecord]:
        """Yield one PolygonRecord per shape in every Kaggle train/val/test JSON.

        Skips the "Test Disease Severity Level" subtree (not for crops).
        Sorted order: by (split, json_path, shape_index) — stable across runs.
        """
        splits = [("train", disease_root / "train"),
                  ("val",   disease_root / "val"),
                  ("test",  disease_root / "test")]
        for split, split_root in splits:
            if not split_root.is_dir():
                logging.warning("Kaggle split dir missing: %s", split_root)
                continue
            for json_path in sorted(split_root.rglob("*.json")):
                try:
                    with open(json_path, encoding="utf-8") as fh:
                        doc = json.load(fh)
                except (OSError, json.JSONDecodeError) as e:
                    logging.warning("Skipping malformed Kaggle JSON %s: %s", json_path, e)
                    continue
                shapes = doc.get("shapes", [])
                image_path = json_path.with_suffix(".jpg")
                if not image_path.exists():
                    logging.warning("Kaggle JSON references missing image: %s", image_path)
                    continue
                image_stem = image_path.stem
                for idx, shape in enumerate(shapes):
                    raw_label = shape.get("label", "").strip()
                    canonical = _canonicalize_kaggle_label(raw_label)
                    if canonical is None:
                        logging.warning("Unknown Kaggle class %r in %s shape %d", raw_label, json_path, idx)
                        continue
                    role = CLASS_ROLE.get(canonical)
                    if role is None:
                        logging.warning("No role for canonical class %r (shape %d in %s)", canonical, idx, json_path)
                        continue
                    pts = tuple((float(x), float(y)) for (x, y) in shape.get("points", []))
                    if len(pts) < 3:
                        logging.warning("Polygon has <3 points in %s shape %d — skipping", json_path, idx)
                        continue
                    yield PolygonRecord(
                        source_image_path=image_path,
                        image_stem=image_stem,
                        polygon_idx=idx,
                        source_class_name=raw_label,
                        canonical_class=canonical,
                        points=pts,
                        split=split,
                        role=role,
                    )

    def _canonicalize_kaggle_label(raw: str) -> str | None:
        """Kaggle labels are already in canonical form EXCEPT casing/whitespace.

        Mirror the behavior in src/manifest/scanners.scan_kaggle_disease to avoid drift.
        """
        norm = raw.lower().strip().replace(" ", "_")
        return norm if norm in CLASS_ROLE else None

    def polygon_to_padded_bbox(
        points: tuple[tuple[float, float], ...],
        image_width: int,
        image_height: int,
        padding_ratio: float = PADDING_RATIO,
    ) -> tuple[int, int, int, int]:
        """Return (left, top, right, bottom) pixel bbox with symmetric padding.

        Pad = padding_ratio * max(bbox_w, bbox_h) applied to each of the 4 sides,
        clamped to [0, image_w] x [0, image_h]. Returns integer pixels for PIL.crop().
        """
        xs = [p[0] for p in points]
        ys = [p[1] for p in points]
        x0, x1 = min(xs), max(xs)
        y0, y1 = min(ys), max(ys)
        bw, bh = x1 - x0, y1 - y0
        pad = padding_ratio * max(bw, bh)
        left   = max(0,             int(round(x0 - pad)))
        top    = max(0,             int(round(y0 - pad)))
        right  = min(image_width,   int(round(x1 + pad)))
        bottom = min(image_height,  int(round(y1 + pad)))
        # Guard against degenerate bboxes
        if right - left < 2 or bottom - top < 2:
            raise ValueError(f"Degenerate bbox after padding: ({left},{top},{right},{bottom})")
        return (left, top, right, bottom)

    def render_crop(
        record: PolygonRecord,
        out_dir: Path,
    ) -> Path | None:
        """Render one padded-bbox crop to disk. Idempotent.

        Output path: out_dir / <image_stem>__<polygon_idx:03d>.jpg
        Returns the written path, or None if the polygon was degenerate (logged, skipped).
        """
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / f"{record.image_stem}__{record.polygon_idx:03d}.jpg"
        try:
            with Image.open(record.source_image_path) as img:
                w, h = img.size
                try:
                    bbox = polygon_to_padded_bbox(record.points, w, h)
                except ValueError as e:
                    logging.warning("Degenerate polygon in %s shape %d: %s", record.source_image_path, record.polygon_idx, e)
                    return None
                crop = img.crop(bbox).convert("RGB")
                tmp = out_path.with_suffix(out_path.suffix + ".tmp")
                crop.save(
                    tmp,
                    format="JPEG",
                    quality=JPEG_QUALITY,
                    optimize=False,
                    progressive=False,
                    subsampling=JPEG_SUBSAMPLING,
                )
                import os
                os.replace(tmp, out_path)
        except OSError as e:
            logging.warning("Failed to render crop for %s shape %d: %s", record.source_image_path, record.polygon_idx, e)
            return None
        return out_path

    def out_dir_for(crops_root: Path, record: PolygonRecord) -> Path:
        """Route a record to its output directory based on role (fruit vs leaf)."""
        if record.role == "leaf":
            return crops_root / "_leaves_excluded" / record.split / record.canonical_class
        return crops_root / record.split / record.canonical_class
    ```

       #!/usr/bin/env python3
       """Materialize data/disease_crops/ from Kaggle Afzaal LabelMe polygons.

       Deterministic: same data/disease/ bytes -> byte-identical data/disease_crops/
       contents, every run. Idempotent: overwrites existing crops atomically.

       Usage:
           python scripts/build_disease_crops.py [--disease-root PATH] [--crops-root PATH] [--quiet]
       """
       from __future__ import annotations
       import argparse, logging, sys
       from collections import Counter
       from pathlib import Path

       REPO_ROOT = Path(__file__).resolve().parent.parent
       sys.path.insert(0, str(REPO_ROOT))

       from src.manifest.crops import iter_kaggle_polygons, render_crop, out_dir_for

       def build(disease_root: Path, crops_root: Path) -> Counter:
           counts: Counter = Counter()
           for record in iter_kaggle_polygons(disease_root):
               dest_dir = out_dir_for(crops_root, record)
               written = render_crop(record, dest_dir)
               if written is None:
                   counts[("skipped", record.canonical_class, record.split)] += 1
                   continue
               counts[(record.role, record.canonical_class, record.split)] += 1
           return counts

       def write_ready_marker(crops_root: Path, counts: Counter) -> None:
           """Write a deterministic READY marker so downstream phases can confirm build success.

           Determinism: sorted keys, no timestamps.
           """
           crops_root.mkdir(parents=True, exist_ok=True)
           lines = ["# data/disease_crops/ READY", ""]
           lines.append("## Fruit classes (training tree)")
           fruit_keys = sorted(k for k in counts if k[0] == "fruit")
           for k in fruit_keys:
               lines.append(f"- {k[1]} / {k[2]}: {counts[k]}")
           lines.append("")
           lines.append("## Leaf classes (excluded from fruit training)")
           leaf_keys = sorted(k for k in counts if k[0] == "leaf")
           for k in leaf_keys:
               lines.append(f"- {k[1]} / {k[2]}: {counts[k]}")
           lines.append("")
           skipped = sorted(k for k in counts if k[0] == "skipped")
           if skipped:
               lines.append("## Skipped (degenerate polygons, missing images, unknown classes)")
               for k in skipped:
                   lines.append(f"- {k[1]} / {k[2]}: {counts[k]}")
               lines.append("")
           marker = crops_root / "READY.md"
           marker.write_text("\n".join(lines) + "\n", encoding="utf-8")

       def main(argv: list[str] | None = None) -> int:
           ap = argparse.ArgumentParser()
           ap.add_argument("--disease-root", type=Path, default=REPO_ROOT / "data/disease")
           ap.add_argument("--crops-root",   type=Path, default=REPO_ROOT / "data/disease_crops")
           ap.add_argument("--quiet", action="store_true")
           args = ap.parse_args(argv)
           logging.basicConfig(
               level=logging.WARNING if args.quiet else logging.INFO,
               format="%(levelname)s %(message)s",
           )
           counts = build(args.disease_root, args.crops_root)
           write_ready_marker(args.crops_root, counts)
           total = sum(counts.values())

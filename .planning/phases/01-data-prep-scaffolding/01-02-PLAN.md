<objective>
Build the authoritative per-image provenance layer for Phase 1. Every image across the 5 verified datasets (Zenodo, Kaggle Afzaal disease, OSF ej5qv, 3 Roboflow mirrors, StrawDI_Db1) gets indexed by sha256 into `data/MANIFEST.json` with license, annotation type, split, and project-schema class attached. A separate dedup audit script reads the manifest, surfaces byte-identical cross-dataset collisions (must reproduce the known `anthracnose_fruit_rot1.jpg` Kaggle↔OSF collision and find any others), and writes `reports/dedup-report.md`. The audit NEVER deletes files — it only reports. Re-running either script on unchanged data produces byte-identical output.

Purpose: Every downstream phase (Phase 2 detector training split construction, Phase 3 disease classifier training, Phase 4 evaluator ground-truth lookup) reads image metadata from this MANIFEST, not by re-walking `data/`. MANIFEST becomes the single source of truth for "which images exist, who owns them, and which ones are duplicates."

Output: `data/MANIFEST.json` (deterministic, re-runnable), `reports/dedup-report.md` (human-readable audit), and an `src/manifest/` library package so Phase 3+ can import ManifestEntry and scanner helpers without re-implementing them.
</objective>

    **src/manifest/hashing.py:**
    ```python
    """Deterministic streaming SHA256 over file bytes. Never reads wall-clock."""
    from __future__ import annotations
    import hashlib
    from pathlib import Path

    _CHUNK = 1 << 20  # 1 MiB

    def sha256_file(path: Path | str) -> str:
        """Return lowercase hex sha256 of the file at path. Streams; no mmap, no timestamps.

        Deterministic: same file bytes -> same digest, always.
        """
        h = hashlib.sha256()
        with open(path, "rb") as fh:
            while True:
                chunk = fh.read(_CHUNK)
                if not chunk:
                    break
                h.update(chunk)
        return h.hexdigest()
    ```

       ```python
       #!/usr/bin/env python3
       """Build data/MANIFEST.json by scanning all 5 verified datasets under data/.

       Deterministic: same inputs -> byte-identical output, every run.
       Idempotent: safe to re-run; overwrites previous MANIFEST.json atomically.

       Usage:
           python scripts/build_manifest.py [--out PATH] [--quiet]
       """
       from __future__ import annotations
       import argparse, json, logging, os, sys, tempfile
       from pathlib import Path

       # Repo-root import shim (scripts are run from repo root in this project).
       REPO_ROOT = Path(__file__).resolve().parent.parent
       sys.path.insert(0, str(REPO_ROOT))

       from src.manifest import (
           ManifestEntry,
           SOURCE_LICENSES,
           scan_zenodo,
           scan_kaggle_disease,
           scan_osf_ej5qv,
           scan_roboflow_yolo,
           scan_strawdi,
       )

       def build_manifest(data_root: Path) -> list[ManifestEntry]:
           entries: list[ManifestEntry] = []
           entries += scan_zenodo(data_root / "zenodo")
           entries += scan_kaggle_disease(data_root / "disease")
           entries += scan_osf_ej5qv(data_root / "osf-ej5qv")
           entries += scan_roboflow_yolo(data_root / "roboflow/afzaal-bbox-v4",   "roboflow_afzaal_bbox_v4")
           entries += scan_roboflow_yolo(data_root / "roboflow/matt-lucky-ripeness", "roboflow_matt_lucky")
           entries += scan_roboflow_yolo(data_root / "roboflow/research-proj-disease", "roboflow_research_proj")
           entries += scan_strawdi(data_root / "strawdi")
           # Deterministic global order: sort by (source, original_path).
           entries.sort(key=lambda e: (e["source"], e["original_path"]))
           return entries

       def write_manifest_atomic(entries: list[ManifestEntry], out: Path) -> None:
           out.parent.mkdir(parents=True, exist_ok=True)
           tmp = out.with_suffix(out.suffix + ".tmp")
           with open(tmp, "w", encoding="utf-8") as fh:
               json.dump(entries, fh, indent=2, sort_keys=True, separators=(",", ": "))
               fh.write("\n")
           os.replace(tmp, out)
       ```

       ```python
       """Audit byte-identical image collisions across datasets using MANIFEST.json.

       Input:  data/MANIFEST.json
       Output: reports/dedup-report.md  (human-readable; does NOT delete any files)

       Determinism: sorted by sha256 then by (source, original_path). Two runs on
       the same MANIFEST produce byte-identical reports.
       """
       from __future__ import annotations
       import argparse, json, sys
       from collections import defaultdict
       from pathlib import Path

       REPO_ROOT = Path(__file__).resolve().parent.parent

       def audit(manifest_path: Path) -> tuple[list[dict], dict]:
           with open(manifest_path, encoding="utf-8") as fh:
               entries = json.load(fh)
           # Group by sha256
           by_hash: dict[str, list[dict]] = defaultdict(list)
           for e in entries:
               by_hash[e["sha256"]].append(e)
           # Collisions = hashes with entries from 2+ distinct sources
           collisions = []
           for h, group in by_hash.items():
               sources = {e["source"] for e in group}
               if len(sources) >= 2:
                   collisions.append({
                       "sha256": h,
                       "copies": sorted(group, key=lambda e: (e["source"], e["original_path"])),
                       "sources": sorted(sources),
                   })
           collisions.sort(key=lambda c: (tuple(c["sources"]), c["sha256"]))
           # Per-source pair counts
           pair_counts: dict[tuple[str, str], int] = defaultdict(int)
           for c in collisions:
               srcs = c["sources"]
               for i, a in enumerate(srcs):
                   for b in srcs[i+1:]:
                       pair_counts[(a, b)] += 1
           summary = {
               "total_entries": len(entries),
               "unique_hashes": len(by_hash),
               "colliding_hashes": len(collisions),
               "pair_counts": dict(sorted(pair_counts.items())),
           }
           return collisions, summary

           lines: list[str] = []
           lines.append("# Dedup Audit Report")
           lines.append("")

           lines.append("")
           lines.append(f"- Total manifest entries: **{summary['total_entries']}**")

           lines.append("")
           lines.append("## Cross-dataset pair collision counts")
           lines.append("")
           lines.append("| Source A | Source B | Collisions |")
           lines.append("|---|---|---|")

           lines.append("")

           lines.append("")
           # Collision detail — first 100 entries, to bound report size
           lines.append("## Collisions (up to first 100 by sorted order)")
           lines.append("")
           if not collisions:
               lines.append("*(none detected)*")
           else:
       ```
